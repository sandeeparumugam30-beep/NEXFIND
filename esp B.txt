
#include <ESP8266WiFi.h>
#include <WiFiUdp.h>

// NEXFIND - ESP B Transmitter

const char* ssid = "YOUR_WIFI_NAME";
const char* password = "YOUR_WIFI_PASSWORD";

WiFiUDP udp;

const unsigned int localPort = 4210;
const unsigned int receiverPort = 4211;

IPAddress receiverIP(192, 168, 1, 255);

unsigned long previousTime = 0;
const unsigned long sendInterval = 1000;

void setup()
{
    Serial.begin(115200);

    Serial.println();
    Serial.println("NEXFIND PART B STARTING");

    WiFi.mode(WIFI_STA);
    WiFi.begin(ssid, password);

    Serial.print("Connecting to Wi-Fi");

    while (WiFi.status() != WL_CONNECTED)
    {
        delay(500);
        Serial.print(".");
    }

    Serial.println();
    Serial.println("Wi-Fi connected");
    Serial.print("ESP B IP: ");
    Serial.println(WiFi.localIP());

    udp.begin(localPort);

    Serial.println("NEXFIND PART B READY");
}

void loop()
{
    if (WiFi.status() != WL_CONNECTED)
    {
        return;
    }

    unsigned long currentTime = millis();

    if (currentTime - previousTime >= sendInterval)
    {
        previousTime = currentTime;

        udp.beginPacket(receiverIP, receiverPort);
        udp.print("NEXFIND_PART_B_ACTIVE");
        udp.endPacket();

        Serial.println("Message sent to ESP A");
    }
}
